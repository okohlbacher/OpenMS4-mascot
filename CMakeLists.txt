cmake_minimum_required(VERSION 3.24)
project(OpenMSMascot VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
include(cmake/OpenMS4Dependencies.cmake)
if(BUILD_TESTING)
  openms4_find_core(COMPONENTS TestSupport)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
else()
  openms4_find_core()
endif()
openms4_find_package(OpenMSCLI)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
add_library(tool_backend STATIC
  source/FORMAT/MascotRemoteQuery.cpp
)
target_include_directories(tool_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(tool_backend PUBLIC cxx_std_23)
target_link_libraries(tool_backend PUBLIC OpenMS::Core)
target_link_libraries(tool_backend PRIVATE CURL::libcurl)

include(cmake/OpenMS4Tools.cmake)
openms4_load_tools(package_tools)
foreach(tool IN LISTS package_tools)
  add_executable(${tool} src/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE tool_backend OpenMS::CLI)
endforeach()
openms4_install_tools(mascot bin ${package_tools})
if(BUILD_TESTING)
  set(CF_OPENMS_TEST_DATA_PATH "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/")
  configure_file(tests/test_config.h.in generated/OpenMS/test_config.h @ONLY)
  file(GLOB class_tests CONFIGURE_DEPENDS tests/source/*_test.cpp)
  foreach(source IN LISTS class_tests)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} ${source} ${OpenMS_TEST_SUPPORT_SOURCE})
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(${name} PRIVATE tool_backend OpenMS::TestFramework)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()

if(BUILD_TESTING)
  add_executable(MascotRemoteQuery_transport_driver tests/mascot_transport_driver.cpp)
  target_link_libraries(MascotRemoteQuery_transport_driver PRIVATE tool_backend)
  add_test(NAME MascotRemoteQuery_transport COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_transport.py $<TARGET_FILE:MascotRemoteQuery_transport_driver>)
  set_tests_properties(MascotRemoteQuery_transport PROPERTIES TIMEOUT 60)
endif()
